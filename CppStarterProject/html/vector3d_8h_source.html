329inline Vector3D Vector3D::getNormalized() const noexcept {
330 float len = length();
331 if (len > 0.0f) {
332 return {x_ / len, y_ / len, z_ / len};
333 } else {
334 return Vector3D::forward();
335 }
336}

338inline Vector3D& Vector3D::normalize() noexcept {
339 auto len = length();
340 if (len > 0.0f) {
341 return operator/=(len);
342 }
343
344 return *this = forward();
345}

347inline constexpr bool Vector3D::isNormalized() const noexcept {
