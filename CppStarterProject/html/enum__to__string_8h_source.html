32// Unfortunately, std::alnum is not constexpr, so we must provide our own implementation.
33// See https://www.open-std.org/jtc1/sc22/wg21/docs/papers/2014/n4203.html for a standard
34// library proposal that hasn't appeared to go anywhere thus far.
35//
39[[nodiscard]] constexpr bool isalnum(char c) noexcept {
40 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9');
41}

60template <typename Enum, Enum EnumValue>
61[[nodiscard]] constexpr std::string_view stringify() noexcept {
62#ifdef __GNUC__
63 std::string_view name = std::string_view{__PRETTY_FUNCTION__};
64#else
65 std::string_view name = std::string_view{__FUNCSIG__};
66#endif
67
68 // Iterate backwards until we hit a non-isalnum character, at which point we've hit a scoping
69 // operator or something similar (in any case, the stop of the enum string).
70 // TODO(C++20) - Use constexpr find_if instead.
71 for (auto i = name.size() - enums::details::offset; i > 0; --i) {
72 if (!enums::details::isalnum(name[i - 1]) && name[i - 1] != '_') {
73 return name.substr(i, name.size() - i - enums::details::offset);
74 }
75 }
76
77 return name.substr(0, name.size() - enums::details::offset);
78}

87template <typename Enum, int Value = -enums::details::MAX_VALUE,
88 int Max = enums::details::MAX_VALUE>
89[[nodiscard]] constexpr std::string_view to_str(int value) noexcept {
90 // If we've matched the requested parameter, return its stringified version.
91 if (value == Value) {
92 return stringify<Enum, static_cast<Enum>(Value)>();
93 }
94 if constexpr (Value + 1 == Max) {
95 return "";
96 } else { // MSVC seems to require an else branch with if-constexpr here. Not sure why.
97 return to_str<Enum, Value + 1, Max>(value); // Recursively increment.
98 }
99}

110template <typename Enum, int Value, int Max>
111[[nodiscard]] constexpr int from_str(const std::string_view& value, int default_value) noexcept {
112 if (value == stringify<Enum, static_cast<Enum>(Value)>()) {
113 return Value;
114 }
115 if constexpr (Value + 1 == Max) {
116 return default_value;
117 } else {
118 return from_str<Enum, Value + 1, Max>(value, default_value);
119 }
120}
