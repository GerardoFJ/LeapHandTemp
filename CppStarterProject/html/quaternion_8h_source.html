477inline Quaternion::Quaternion(const Vector3D& axis, float theta) noexcept {
478 const float len = axis.length();
479 r_ = cosf(theta / 2.0f);
480 const float sin_theta_over_2 = sinf(theta / 2.0f);
481 i_ = (axis.x_ / len) * sin_theta_over_2;
482 j_ = (axis.y_ / len) * sin_theta_over_2;
483 k_ = (axis.z_ / len) * sin_theta_over_2;
484}

486// NOLINTNEXTLINE(cppcoreguidelines-pro-type-member-init,hicpp-member-init)
487inline Quaternion::Quaternion(float z_rad, float y_rad, float x_rad) noexcept {
488 float cosx = cosf(x_rad * 0.5f);
489 float sinx = sinf(x_rad * 0.5f);
490 float cosy = cosf(y_rad * 0.5f);
491 float siny = sinf(y_rad * 0.5f);
492 float cosz = cosf(z_rad * 0.5f);
493 float sinz = sinf(z_rad * 0.5f);
494
495 r_ = cosx * cosy * cosz + sinx * siny * sinz;
496 i_ = sinx * cosy * cosz - cosx * siny * sinz;
497 j_ = cosx * siny * cosz + sinx * cosy * sinz;
498 k_ = cosx * cosy * sinz - sinx * siny * cosz;
499}

501inline Vector3D Quaternion::toEuler() const noexcept {
502 float jsqr = j_ * j_;
503
504 float f0 = 2.0f * (r_ * i_ + j_ * k_);
505 float f1 = 1.0f - 2.0f * (i_ * i_ + jsqr);
506 float x_rad = atan2(f0, f1);
507
508 float f2 = 2.0f * (r_ * j_ - k_ * i_);
509 f2 = f2 > 1.0f ? 1.0f : f2;
510 f2 = f2 < -1.0f ? -1.0f : f2;
511 float y_rad = asin(f2);
512
513 float f3 = 2.0f * (r_ * k_ + i_ * j_);
514 float f4 = 1.0f - 2.0f * (jsqr + k_ * k_);
515 float z_rad = atan2(f3, f4);
516
517 return {x_rad, y_rad, z_rad};
518}

575inline constexpr Quaternion Quaternion::product(const Quaternion& other) const noexcept {
576 return {r_ * other.r_ - i_ * other.i_ - j_ * other.j_ - k_ * other.k_,
577 r_ * other.i_ + i_ * other.r_ + j_ * other.k_ - k_ * other.j_,
578 r_ * other.j_ - i_ * other.k_ + j_ * other.r_ + k_ * other.i_,
579 r_ * other.k_ + i_ * other.j_ - j_ * other.i_ + k_ * other.r_};

596inline Vector3D Quaternion::rotate(const Vector3D& other) const noexcept {
597 Quaternion normalized = this->getNormalized();
598 Quaternion intermediate = normalized.product(Quaternion(0, other.x_, other.y_, other.z_));
599 Quaternion result = intermediate.product(normalized.getInverse());
600 return {result.i_, result.j_, result.k_};
601}
