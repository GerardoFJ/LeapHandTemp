60static EDegreeOfFreedom makeDegreeOfFreedomFromDomainAndAxis(EDofDomain domain, EDofAxis axis) {

152static const void getOrthonormalDirectionsFromDegreeOfFreedom(
153 EDegreeOfFreedom degree_of_freedom, FVector& first, FVector& second) {
154 switch (degree_of_freedom) {
155 case EDegreeOfFreedom::X_LIN: {
156 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::Y_LIN);
157 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::Z_LIN);
158 return;
159 }
160 case EDegreeOfFreedom::Y_LIN: {
161 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::Z_LIN);
162 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::X_LIN);
163 return;
164 }
165 case EDegreeOfFreedom::Z_LIN: {
166 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::X_LIN);
167 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::Y_LIN);
168 return;
169 }
170 case EDegreeOfFreedom::X_ANG: {
171 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::Y_ANG);
172 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::Z_ANG);
173 return;
174 }
175 case EDegreeOfFreedom::Y_ANG: {
176 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::Z_ANG);
177 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::X_ANG);
178 return;
179 }
180 case EDegreeOfFreedom::Z_ANG: {
181 first = directionOfDegreeOfFreedom(EDegreeOfFreedom::X_ANG);
182 second = directionOfDegreeOfFreedom(EDegreeOfFreedom::Y_ANG);
183 return;
184 }
185 default: {
186 first = FVector::OneVector;
187 second = FVector::OneVector;
188 return;
189 }
190 }
191}

201static const FVector directionOfDegreeOfFreedom(FQuat& rotation,
202 EDegreeOfFreedom degree_of_freedom) {
203 switch (degree_of_freedom) {
204 case EDegreeOfFreedom::X_LIN:
205 return rotation.GetForwardVector();
206 case EDegreeOfFreedom::Y_LIN:
207 return rotation.GetRightVector();
208 case EDegreeOfFreedom::Z_LIN:
209 return rotation.GetUpVector();
210 case EDegreeOfFreedom::X_ANG:
211 return -rotation.GetForwardVector();
212 case EDegreeOfFreedom::Y_ANG:
213 return -rotation.GetRightVector();
214 case EDegreeOfFreedom::Z_ANG:
215 return rotation.GetUpVector();
216 default:
217 return FVector::OneVector;
218 }
219}

230static const void getOrthonormalDirectionsFromDegreeOfFreedom(
