332 static inline FHandPhysicsTargets interpolate(const FHandPhysicsTargets& a,
333 const FHandPhysicsTargets& b, float alpha) {
334 FHandPhysicsTargets c;
335 c.w_middle1_pos_cm = UKismetMathLibrary::VLerp(a.w_middle1_pos_cm, b.w_middle1_pos_cm, alpha);
336 c.w_middle1_orient = FQuat::Slerp(a.w_middle1_orient, b.w_middle1_orient, alpha);
337 int min_num = FMath::Min(a.l_joint_orients.Num(), b.l_joint_orients.Num());
338 c.l_joint_orients.SetNum(min_num);
339 for (int i = 0; i < min_num; i++) {
340 c.l_joint_orients[i] = FQuat::Slerp(a.l_joint_orients[i], b.l_joint_orients[i], alpha);
341 }
342 return c;
343 }

363struct FHandPhysicsState {
364
365
367
368 TArray<FRigidBodyState> w_body_states;
369
371
372 FHandPhysicsTargets targets;
373
375
376 TArray<FObjectPhysicsState> w_object_states;
377
379
380 TArray<FConstraintPhysicsState> constraint_states;
381
389 static inline FHandPhysicsState interpolate(const FHandPhysicsState& a,
390 const FHandPhysicsState& b, float alpha) {
391 FHandPhysicsState c;
392 int min_num = FMath::Min(a.w_body_states.Num(), b.w_body_states.Num());
393 c.w_body_states.SetNum(min_num);
394 for (int i = 0; i < min_num; i++) {
395 c.w_body_states[i] =
396 interpolateRigidBodyState(a.w_body_states[i], b.w_body_states[i], alpha);
397 }
398
399 c.targets = FHandPhysicsTargets::interpolate(a.targets, b.targets, alpha);
400
401 c.w_object_states.SetNum(a.w_object_states.Num());
402 int c_i = 0;
403 for (int a_i = 0; a_i < a.w_object_states.Num(); a_i++) {
404 for (int b_i = 0; b_i < b.w_object_states.Num(); b_i++) {
405 if (a.w_object_states[a_i].component == b.w_object_states[b_i].component &&
406 a.w_object_states[a_i].body_index == b.w_object_states[b_i].body_index) {
407 c.w_object_states[c_i].component = a.w_object_states[a_i].component;
408 c.w_object_states[c_i].body_index = a.w_object_states[a_i].body_index;
409 c.w_object_states[c_i].state = interpolateRigidBodyState(a.w_object_states[a_i].state,
410 b.w_object_states[b_i].state, alpha);
411 c_i++;
412 break;
413 }
414 }
415 }
416 c.w_object_states.SetNum(c_i, true);
417
418 // TODO: Figure out how to interpolate constraint states.
419 c.constraint_states = a.constraint_states;
420 return c;
421 }
422};
