53class TotalRotation {
54
55public:
56
58 TotalRotation() : revolution_(0), partial_angle_(0.f) {}
59
64 TotalRotation(int revolution, float partial_rot) : revolution_(revolution),
65 partial_angle_(partial_rot) {}
66
70 explicit TotalRotation(float total_rot) {
71 const float offset = total_rot < 0 ? -DEG_PER_REV / 2 : DEG_PER_REV / 2;
72 revolution_ = (int)truncf((total_rot + offset) / DEG_PER_REV);
73 partial_angle_ = total_rot - revolution_ * DEG_PER_REV;
74 }
75
79 float getTotalRotation() const {
80 return partial_angle_ + revolution_ * DEG_PER_REV;
81 }
82
86 FString ToString() const {
87 return FString::Printf(TEXT("Total rotation = %f, revolution = %d, partial angle = %f"),
88 getTotalRotation(), revolution_, partial_angle_);
89 }
90
95 int revolution_;
96
100 float partial_angle_;
101};
